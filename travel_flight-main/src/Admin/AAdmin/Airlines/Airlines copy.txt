import React, { useEffect, useState } from "react";
import { useParams } from "react-router-dom";
import useApiService from 'src/services/useApiService';
import Notification from 'src/hooks/Notification';

const initialState = {
  airline_id: "",
  name: "",
  country: "",
  code: "",
}
const Airlines = (props) => {
  const { id } = useParams();
  const numericId = id ? parseInt(id, 10) : null;
  const [formData, setFormData] = useState(initialState);
  const [notificationMessage, setNotificationMessage] = useState("");

  const route = "/airlines";
  const { data, loading, setLoading, error, fetchData, createData, updateData } = useApiService();

  useEffect(() => {
    // Fetch data when the component loads, only in Edit mode
    if (props.page === "Edit" && numericId) {
      initialState.id = "";
      const fetchGetData = async () => {
        try {
          await fetchData(route, numericId);
        } catch (err) {
          console.error("Error fetching flight data:", err);
        }
      };
      fetchGetData();
    }
  }, [props.page, numericId, route]);
  
  // Sync formData with data when data is available
  useEffect(() => {
    if (props.page === "Edit" && data) {
      setFormData((prev) => ({
        ...prev,
        ...data,
      }));
    }
  }, [data, props.page]);

  const handleChange = (e) => {
    const { name, value } = e.target;
    setFormData((prev) => ({
      ...prev,
      [name]: value,
    }));
  };

  const handleSubmit = async (e) => {
    e.preventDefault();
    setLoading(true);
    try {
      if(props.page === "Edit") {
        const updateItem = await updateData(`${route}/${numericId}`, formData);
        setNotificationMessage("Update Successful");
      } else {
        const createdItem = await createData(route, formData);
        setNotificationMessage("Create Successful");
        setFormData(initialState);
      }

    } catch (err) {
      console.error("Error creating item:", err);
    }
    setLoading(false);
    console.log("Airline Data Submitted:", formData);
  };

  return (
    <div
      style={{
        display: "flex",
        alignItems: "center",
        justifyContent: "center",
        height: "100vh",
        background: "linear-gradient(90deg, #f7f7f7, #e9f7f9)",
      }}
    >
      <form
        onSubmit={handleSubmit}
        style={{
          background: "#fff",
          padding: "20px",
          borderRadius: "8px",
          boxShadow: "0px 4px 10px rgba(0, 0, 0, 0.1)",
          width: "400px",
        }}
      >
        <h3 style={{ textAlign: "center", marginBottom: "15px" }}>Airline Details</h3>

        {(props.page === 'Edit') && 
         <div style={{ marginBottom: "15px" }}>
          <label style={{ display: "block", marginBottom: "5px" }}>Airlines ID:</label>
          <input
            type="number"
            name="id"
            value={formData.id || ""}
            onChange={handleChange}
            required
            disabled
            style={{
              width: "100%",
              padding: "8px",
              border: "1px solid #ccc",
              borderRadius: "5px",
            }}
          />
        </div> }

        {["name", "country", "code"].map((field) => (
          <div key={field} style={{ marginBottom: "15px" }}>
            <label style={{ display: "block", marginBottom: "5px" }}>{field.replace("_", " ").toUpperCase()}:</label>
            <input
              type="text"
              name={field}
              value={formData[field]}
              onChange={handleChange}
              required
              style={{
                width: "100%",
                padding: "8px",
                border: "1px solid #ccc",
                borderRadius: "5px",
              }}
            />
          </div>
        ))}
        <button
          type="submit"
          style={{
            width: "100%",
            padding: "10px",
            backgroundColor: "#007BFF",
            color: "#fff",
            border: "none",
            borderRadius: "5px",
            cursor: "pointer",
          }}          
          disabled={loading}
        >
          {loading ? "Submitting..." : (props.page === "Edit") ? "Edit": "Submit"}
        </button>
      </form>
      
      {notificationMessage && <Notification message={notificationMessage} />}
    </div>
  );
};

export default Airlines;
