import React, { useEffect, useState } from "react";
import { useParams } from "react-router-dom";
import useApiService from "src/services/useApiService";
import Notification from "src/hooks/Notification";

const initialState = {
    reservation_id: "",
    first_name: "",
    last_name: "",
    passport_number: "",
    nationality: "",
};

const Passengers = (props) => {
    const { id } = useParams();
    const numericId = id ? parseInt(id, 10) : null;
    const [formData, setFormData] = useState(initialState);
    const [notificationMessage, setNotificationMessage] = useState("");
    const [reservations, setReservations] = useState([]);
    const [filteredReservations, setFilteredReservations] = useState([]);

    const route = {
        passenger: "/passengers",
        country: "/countries",
        reservation: "/reservations",
    };

    const {
        data: passengerData,
        loading: passengerLoading,
        setLoading: setPassengerLoading,
        error: passengerError,
        fetchData: fetchPassengerData,
        createData: createPassengerData,
        updateData: updatePassengerData,
    } = useApiService();

    const { data: countryData, fetchData: fetchCountryData } = useApiService();

    props.page === "Edit" && (initialState.id = 0);

    useEffect(() => {
        // Fetch data when the component loads, only in Edit mode
        if (props.page === "Edit" && numericId) {
            initialState.id = "";
            const fetchGetData = async () => {
                try {
                    await fetchPassengerData(route.passenger, numericId);
                } catch (err) {
                    console.error("Error fetching data:", err);
                }
            };
            fetchGetData();
        }

        fetchCountryData(route.country);

        // Fetch all reservations for autocomplete
        const fetchReservations = async () => {
            try {
                const res = await fetchPassengerData(route.reservation);
                setReservations(res);
            } catch (err) {
                console.error("Error fetching reservations:", err);
            }
        };

        fetchReservations();
    }, [props.page, numericId, route.passenger, route.country, route.reservation]);

    // Sync formData with data when data is available
    useEffect(() => {
        if (props.page === "Edit" && passengerData) {
            setFormData((prev) => ({
                ...prev,
                ...passengerData,
            }));
        }
    }, [passengerData, props.page]);

    const handleChange = (e) => {
        const { name, value } = e.target;

        setFormData((prev) => ({
            ...prev,
            [name]: value,
        }));

        // Filter reservations for suggestions
        if (name === "reservation_id") {
            const filtered = reservations.filter((reservation) =>
                reservation.id.toString().startsWith(value)
            );
            setFilteredReservations(filtered);
        }
    };

    const handleSelect = (reservationId) => {
        setFormData((prev) => ({
            ...prev,
            reservation_id: reservationId,
        }));
        setFilteredReservations([]); // Clear suggestions
    };

    const handleSubmit = async (e) => {
        e.preventDefault();
        setPassengerLoading(true);
        try {
            if (props.page === "Edit") {
                const updateItem = await updatePassengerData(
                    `${route.passenger}/${numericId}`,
                    formData
                );
                setNotificationMessage("Update Successful");
            } else {
                const createdItem = await createPassengerData(
                    route.passenger,
                    formData
                );
                setNotificationMessage("Create Successful");
                setFormData(initialState);
            }
        } catch (err) {
            console.error("Error creating item:", err);
        }
        setPassengerLoading(false);
    };

    return (
        <div
            style={{
                display: "flex",
                alignItems: "center",
                justifyContent: "center",
                height: "100vh",
                background: "linear-gradient(90deg, #f7f7f7, #e9f7f9)",
            }}
        >
            <form
                onSubmit={handleSubmit}
                style={{
                    background: "#fff",
                    padding: "20px",
                    borderRadius: "8px",
                    boxShadow: "0px 4px 10px rgba(0, 0, 0, 0.1)",
                    width: "400px",
                }}
            >
                <h3 style={{ textAlign: "center", marginBottom: "15px" }}>
                    Passenger Details
                </h3>

                {props.page === "Edit"}
                <div style={{ marginBottom: "15px" }}>
                    <label style={{ display: "block", marginBottom: "5px" }}>
                        Reservation ID:
                    </label>
                    <input
                        type="text"
                        name="reservation_id"
                        value={formData.reservation_id}
                        onChange={handleChange}
                        required
                        style={{
                            width: "100%",
                            padding: "8px",
                            border: "1px solid #ccc",
                            borderRadius: "5px",
                        }}
                        autoComplete="off"
                    />
                    {/* Suggestion Dropdown */}
                    {filteredReservations.length > 0 && (
                        <ul
                            style={{
                                listStyle: "none",
                                padding: "5px",
                                margin: "0",
                                border: "1px solid #ccc",
                                borderRadius: "5px",
                                maxHeight: "150px",
                                overflowY: "auto",
                                backgroundColor: "#fff",
                            }}
                        >
                            {filteredReservations.map((reservation) => (
                                <li
                                    key={reservation.id}
                                    onClick={() => handleSelect(reservation.id)}
                                    style={{
                                        padding: "8px",
                                        cursor: "pointer",
                                        borderBottom: "1px solid #eee",
                                    }}
                                >
                                    {reservation.id} - {reservation.flight_id} -{" "}
                                    {reservation.reservation_date}
                                </li>
                            ))}
                        </ul>
                    )}
                </div>

                <div style={{ marginBottom: "15px" }}>
                    <label style={{ display: "block", marginBottom: "5px" }}>
                        Country:
                    </label>
                    <select
                        name="nationality"
                        value={formData.nationality || ""}
                        onChange={handleChange}
                        required
                        style={{
                            width: "100%",
                            padding: "8px",
                            border: "1px solid #ccc",
                            borderRadius: "5px",
                        }}
                    >
                        <option value="" disabled>
                            --- Select Country ---
                        </option>
                        {Array.isArray(countryData) &&
                        countryData.length > 0 ? (
                            countryData.map((item) => (
                                <option key={item.id} value={item.id}>
                                    {item.country_name}
                                </option>
                            ))
                        ) : (
                            <option disabled>Loading countries...</option>
                        )}
                    </select>
                </div>

                <button
                    type="submit"
                    style={{
                        width: "100%",
                        padding: "10px",
                        backgroundColor: "#007BFF",
                        color: "#fff",
                        border: "none",
                        borderRadius: "5px",
                        cursor: "pointer",
                    }}
                >
                    Submit
                </button>
            </form>

            {notificationMessage && (
                <Notification message={notificationMessage} />
            )}
        </div>
    );
};

export default Passengers;
